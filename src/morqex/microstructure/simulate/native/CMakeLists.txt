cmake_minimum_required(VERSION 3.24)
project(gem_echemamr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GEM_ECHEM_CASE "models/Microstructure" CACHE STRING "Upstream problem model")
set(AMReX_SPACEDIM 3 CACHE STRING "3D")
set(AMReX_MPI OFF CACHE BOOL "MPI")
set(AMReX_OMP OFF CACHE BOOL "OpenMP")
set(AMReX_FORTRAN OFF CACHE BOOL "No Fortran")
set(AMReX_PARTICLES OFF CACHE BOOL "No particles")
set(AMReX_AMRLEVEL OFF CACHE BOOL "AmrCore only")
set(AMReX_LINEAR_SOLVERS ON CACHE BOOL "Multigrid")
set(AMReX_LINEAR_SOLVERS_EM OFF CACHE BOOL "echemAMR has no electromagnetic nodal solve")
set(AMReX_EB OFF CACHE BOOL "Immersed interface")
set(AMReX_GPU_BACKEND "NONE" CACHE STRING "NONE or CUDA")
set(AMReX_CUDA_FASTMATH OFF CACHE BOOL "Preserve numerical precision")
include(FetchContent)
FetchContent_Declare(amrex
  URL https://codeload.github.com/AMReX-Codes/amrex/tar.gz/9ed35ab0e58462f951528462376cc379ba834826
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(amrex)
# echemAMR uses cell-centered ABec and tensor operators. Unused nodal fluid
# operators trigger pathological MSVC optimization and needlessly enlarge builds.
get_target_property(gem_amrex_sources amrex_3d SOURCES)
list(FILTER gem_amrex_sources EXCLUDE REGEX "AMReX_MLNode(Laplacian|ABecLaplacian).*\\.cpp$")
set_property(TARGET amrex_3d PROPERTY SOURCES "${gem_amrex_sources}")
if(MSVC)
  # Pinned upstream declares static const then defines constexpr. MSVC C2370;
  # retain const values and inline ODR semantics without changing arithmetic.
  foreach(header AMReX_IntVect.H AMReX_RealVect.H)
    set(path "${amrex_SOURCE_DIR}/Src/Base/${header}")
    file(READ "${path}" content)
    string(REPLACE "inline constexpr const IntVectND<dim>" "inline const IntVectND<dim>" content "${content}")
    string(REPLACE "inline constexpr const RealVectND<dim>" "inline const RealVectND<dim>" content "${content}")
    file(WRITE "${path}" "${content}")
  endforeach()
endif()
find_package(pybind11 CONFIG REQUIRED)
file(GLOB engine CONFIGURE_DEPENDS "upstream/Source/*.cpp")
set(case_dir "${CMAKE_CURRENT_SOURCE_DIR}/upstream/${GEM_ECHEM_CASE}")
file(GLOB chemistry CONFIGURE_DEPENDS "${case_dir}/*.cpp")
pybind11_add_module(_gem_echemamr bindings.cpp ${engine} ${chemistry})
set_source_files_properties(upstream/Source/main.cpp PROPERTIES COMPILE_DEFINITIONS "main=gem_echem_main")
target_include_directories(_gem_echemamr PRIVATE upstream/Source "${case_dir}")
target_link_libraries(_gem_echemamr PRIVATE AMReX::amrex)
if(MSVC)
  target_compile_definitions(_gem_echemamr PRIVATE _USE_MATH_DEFINES NOMINMAX)
  target_compile_options(_gem_echemamr PRIVATE /bigobj /EHsc)
endif()
if(AMReX_GPU_BACKEND STREQUAL "CUDA")
  enable_language(CUDA)
  setup_target_for_cuda_compilation(_gem_echemamr)
endif()
install(TARGETS _gem_echemamr LIBRARY DESTINATION . RUNTIME DESTINATION .)
