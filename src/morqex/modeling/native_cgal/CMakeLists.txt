cmake_minimum_required(VERSION 3.16)
project(morqex_cgal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MORQEX_CGAL_DEPENDENCY_PREFIX "" CACHE PATH "External Boost/Eigen/GMP/MPFR development prefix")
set(MORQEX_PYTHON_EXTENSION OFF CACHE BOOL "Expose an optional Python extension initializer")
set(MORQEX_SOURCE_ID "untracked" CACHE STRING "MORQEX adapter and upstream source identity")
set(_include "${MORQEX_CGAL_DEPENDENCY_PREFIX}/include")
find_path(MQ_BOOST boost/version.hpp HINTS "${_include}")
find_path(MQ_EIGEN Eigen/Core HINTS "${_include}/eigen3" PATH_SUFFIXES eigen3)
find_path(MQ_GMP_INCLUDE gmp.h HINTS "${_include}" "${_include}/x86_64-linux-gnu" PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu)
find_path(MQ_MPFR_INCLUDE mpfr.h HINTS "${_include}")
if(UNIX AND NOT APPLE)
  # A downloaded development prefix may contain dangling .so links and a
  # non-PIC .a. Link the shared runtime, never that static archive.
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".so" ".so.10" ".so.6")
endif()
find_library(MQ_GMP NAMES libgmp.so.10 gmp HINTS "${MORQEX_CGAL_DEPENDENCY_PREFIX}/lib" "${MORQEX_CGAL_DEPENDENCY_PREFIX}/lib/x86_64-linux-gnu")
find_library(MQ_MPFR NAMES libmpfr.so.6 mpfr HINTS "${MORQEX_CGAL_DEPENDENCY_PREFIX}/lib" "${MORQEX_CGAL_DEPENDENCY_PREFIX}/lib/x86_64-linux-gnu")
foreach(required MQ_BOOST MQ_EIGEN MQ_GMP_INCLUDE MQ_MPFR_INCLUDE MQ_GMP MQ_MPFR)
  if(NOT ${required})
    message(FATAL_ERROR "Missing ${required}; supply external Boost/Eigen3/GMP/MPFR headers and runtime libraries. See MORQEX geometry tutorial.")
  endif()
endforeach()
add_library(morqex_cgal SHARED bridge.cpp)
target_include_directories(morqex_cgal PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/cgal/include" "${MQ_BOOST}" "${MQ_EIGEN}" "${MQ_GMP_INCLUDE}" "${MQ_MPFR_INCLUDE}")
target_compile_definitions(morqex_cgal PRIVATE CGAL_EIGEN3_ENABLED)
target_compile_definitions(morqex_cgal PRIVATE MORQEX_SOURCE_ID="${MORQEX_SOURCE_ID}")
target_link_libraries(morqex_cgal PRIVATE "${MQ_GMP}" "${MQ_MPFR}")
if(MORQEX_PYTHON_EXTENSION)
  find_package(Python COMPONENTS Development.Module REQUIRED)
  target_link_libraries(morqex_cgal PRIVATE Python::Module)
  target_compile_definitions(morqex_cgal PRIVATE MORQEX_PYTHON_EXTENSION=1)
endif()
set_target_properties(morqex_cgal PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(NOT WIN32)
  set_target_properties(morqex_cgal PROPERTIES PREFIX "lib")
endif()
