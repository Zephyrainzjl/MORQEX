cmake_minimum_required(VERSION 3.18)
project(morqex_mmg LANGUAGES C CXX)
set(BUILD MMG3D CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SCOTCH OFF CACHE STRING "" FORCE)
set(USE_ELAS OFF CACHE STRING "" FORCE)
set(USE_VTK OFF CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH_USE_ORIGIN TRUE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/mmg" mmg)
set_target_properties(libmmg3d_so PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_library(morqex_mmg SHARED bridge.cpp)
target_compile_features(morqex_mmg PRIVATE cxx_std_17)
target_include_directories(morqex_mmg PRIVATE "${CMAKE_BINARY_DIR}/mmg/include")
target_link_libraries(morqex_mmg PRIVATE libmmg3d_so)
add_dependencies(morqex_mmg copy_3d_headers)
set_target_properties(morqex_mmg PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(MORQEX_PYTHON_EXTENSION)
  find_package(Python COMPONENTS Development.Module REQUIRED)
  target_compile_definitions(morqex_mmg PRIVATE MORQEX_PYTHON_EXTENSION=1)
  target_link_libraries(morqex_mmg PRIVATE Python::Module)
endif()
