cmake_minimum_required(VERSION 3.20)
project(morqex_top_damage LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MORQEX_MFEM_FROM_SOURCE "Build the vendored MFEM library" ON)
if(MORQEX_MFEM_FROM_SOURCE)
  set(MFEM_USE_MPI OFF CACHE BOOL "" FORCE)
  set(MFEM_USE_METIS OFF CACHE BOOL "" FORCE)
  set(MFEM_USE_OPENMP OFF CACHE BOOL "" FORCE)
  set(MFEM_USE_CUDA OFF CACHE BOOL "" FORCE)
  set(MFEM_USE_EXCEPTIONS ON CACHE BOOL "" FORCE)
  set(MFEM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(MFEM_ENABLE_MINIAPPS OFF CACHE BOOL "" FORCE)
  set(MFEM_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  add_subdirectory(../../vendor/mfem mfem)
else()
  # The upstream wheel's exported target embeds /usr/lib64/libz.so.
  # Find the actual local library instead of modifying third-party files.
  find_path(MFEM_INCLUDE mfem.hpp PATHS "${MFEM_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(MFEM_LIBRARY mfem PATHS "${MFEM_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  find_package(ZLIB REQUIRED)
  add_library(mfem SHARED IMPORTED)
  set_target_properties(mfem PROPERTIES IMPORTED_LOCATION "${MFEM_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${MFEM_INCLUDE}"
    INTERFACE_LINK_LIBRARIES ZLIB::ZLIB)
endif()
add_library(morqex_phase_field SHARED scalar_form.cpp)
target_link_libraries(morqex_phase_field PRIVATE mfem)
target_compile_options(morqex_phase_field PRIVATE -Wall -Wextra)
