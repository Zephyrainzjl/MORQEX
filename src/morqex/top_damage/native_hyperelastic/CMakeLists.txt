cmake_minimum_required(VERSION 3.20)
project(morqex_hyperelastic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Reuse a hash-verified, already compiled MFEM; no upstream rebuild or downloads.
foreach(required MFEM_SOURCE MFEM_BUILD MFEM_LIBRARY)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
add_library(morqex_hyperelastic SHARED constitutive.cpp)
target_include_directories(morqex_hyperelastic PRIVATE "${MFEM_BUILD}" "${MFEM_SOURCE}")
target_link_libraries(morqex_hyperelastic PRIVATE "${MFEM_LIBRARY}")
target_compile_options(morqex_hyperelastic PRIVATE -Wall -Wextra)
