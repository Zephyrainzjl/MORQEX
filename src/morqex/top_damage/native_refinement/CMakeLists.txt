cmake_minimum_required(VERSION 3.20)
project(morqex_refinement LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
foreach(required MFEM_SOURCE MFEM_BUILD MFEM_LIBRARY)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
add_library(morqex_refinement SHARED refinement.cpp)
target_include_directories(morqex_refinement PRIVATE "${MFEM_BUILD}" "${MFEM_SOURCE}")
target_link_libraries(morqex_refinement PRIVATE "${MFEM_LIBRARY}")
target_compile_options(morqex_refinement PRIVATE -Wall -Wextra)
